Firestore Production Security Rules

Paste into Firebase Console → Firestore Database → Rules tab to enforce server-side access control

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // 1. Inquiries: Public can submit inquiries with validation; only admin can view/manage
    match /inquiries/{inquiryId} {
      allow create: if request.resource.data.name is string && request.resource.data.phone is string;
      allow read, update, delete: if request.auth != null;
    }
    // 2. Portfolio Projects: Public can view, only authenticated admin can create/edit/delete
    match /projects/{projectId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    // 3. Security Settings: STRICTLY locked, never publicly readable or writable
    match /settings/security {
      allow read, write: if request.auth != null;
    }
    // 4. Public Website Content & Branding: Public can read, only admin can write
    match /settings/{settingId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    // 4. Testimonials & Reviews: Public can read, only admin can manage
    match /reviews/{reviewId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
